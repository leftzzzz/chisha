```typescript
import { createContext, useContext, useReducer, ReactNode } from 'react';
import { appReducer, initialState } from './AppReducer';
import type { AppState, AppAction } from '@/types';

interface AppContextType {
  state: AppState;
  dispatch: React.Dispatch<AppAction>;
}

const AppContext = createContext<AppContextType | null>(null);

export function AppProvider({ children }: { children: ReactNode }) {
  const [state, dispatch] = useReducer(appReducer, initialState);

  return (
    <AppContext.Provider value={{ state, dispatch }}>
      {children}
    </AppContext.Provider>
  );
}

export function useAppContext() {
  const context = useContext(AppContext);
  if (!context) {
    throw new Error('useAppContext must be used within AppProvider');
  }
  return context;
}
```

### 2.2 状态 Reducer

```typescript
// context/AppReducer.ts
import type { AppState, AppAction, AppStep } from '@/types';

export const initialState: AppState = {
  step: 'INPUT',
  query: '',
  location: null,
  parsedRequirement: null,
  restaurants: [],
  selectedIndex: null,
  selectedRestaurant: null,
  error: null,
  isLoading: false,
};

export function appReducer(state: AppState, action: AppAction): AppState {
  switch (action.type) {
    case 'SET_QUERY':
      return { ...state, query: action.payload };

    case 'SET_LOCATION':
      return { ...state, location: action.payload };

    case 'START_UNDERSTANDING':
      return {
        ...state,
        step: 'UNDERSTANDING',
        isLoading: true,
        error: null,
      };

    case 'UNDERSTANDING_SUCCESS':
      return {
        ...state,
        step: 'SEARCHING',
        parsedRequirement: action.payload,
      };

    case 'SEARCH_SUCCESS':
      return {
        ...state,
        step: 'READY',
        restaurants: action.payload,
        isLoading: false,
      };

    case 'START_SPIN':
      return { ...state, step: 'SPINNING' };

    case 'SPIN_COMPLETE':
      return {
        ...state,
        step: 'RESULT',
        selectedIndex: action.payload,
        selectedRestaurant: state.restaurants[action.payload],
      };

    case 'REMOVE_RESTAURANT':
      const newRestaurants = state.restaurants.filter(
        (r) => r.id !== action.payload
      );
      return {
        ...state,
        restaurants: newRestaurants,
        step: newRestaurants.length > 0 ? 'READY' : 'ERROR',
        error: newRestaurants.length === 0 ? '没有更多餐厅了' : null,
      };

    case 'SET_ERROR':
      return {
        ...state,
        step: 'ERROR',
        error: action.payload,
        isLoading: false,
      };

    case 'RESET':
      return { ...initialState, location: state.location };

    case 'BACK_TO_READY':
      return {
        ...state,
        step: 'READY',
        selectedIndex: null,
        selectedRestaurant: null,
      };

    default:
      return state;
  }
}
```

### 3.1 位置获取 Hook

```typescript
// hooks/useLocation.ts
import { useState, useEffect, useCallback } from 'react';
import { reverseGeocode } from '@/lib/api';
import type { Location } from '@/types';

interface UseLocationResult {
  location: Location | null;
  isLoading: boolean;
  error: string | null;
  requestLocation: () => void;
  setManualLocation: (address: string) => Promise<void>;
}

export function useLocation(): UseLocationResult {
  const [location, setLocation] = useState<Location | null>(null);
  const [isLoading, setIsLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const requestLocation = useCallback(() => {
    if (!navigator.geolocation) {
      setError('浏览器不支持定位');
      return;
    }

    setIsLoading(true);
    setError(null);

    navigator.geolocation.getCurrentPosition(
      async (position) => {
        const { latitude: lat, longitude: lng } = position.coords;

        try {
          // 逆地理编码获取地址
          const result = await reverseGeocode({ lat, lng });
          setLocation({
            lat,
            lng,
            address: result.formattedAddress,
          });
        } catch {
          // 即使逆编码失败，也使用坐标
          setLocation({ lat, lng });
        }

        setIsLoading(false);
      },
      (err) => {
        setError('无法获取位置，请手动输入');
        setIsLoading(false);
      },
      {
        timeout: 5000,
        enableHighAccuracy: true,
      }
    );
  }, []);

  const setManualLocation = useCallback(async (address: string) => {
    setIsLoading(true);
    setError(null);

    try {
      const result = await geocode({ address });
      setLocation({
        lat: result.location.lat,
        lng: result.location.lng,
        address: result.formattedAddress,
      });
    } catch {
      setError('地址解析失败，请重试');
    }

    setIsLoading(false);
  }, []);

  return { location, isLoading, error, requestLocation, setManualLocation };
}
```

### 3.2 餐厅搜索 Hook

```typescript
// hooks/useRestaurantSearch.ts
import { useCallback } from 'react';
import { useAppContext } from '@/context/AppContext';
import { understand, searchRestaurants } from '@/lib/api';

export function useRestaurantSearch() {
  const { state, dispatch } = useAppContext();

  const search = useCallback(async () => {
    if (!state.query || !state.location) {
      dispatch({ type: 'SET_ERROR', payload: '请输入需求和位置' });
      return;
    }

    try {
      // 1. 开始理解
      dispatch({ type: 'START_UNDERSTANDING' });

      const parsed = await understand({
        query: state.query,
        location: state.location,
      });

      dispatch({ type: 'UNDERSTANDING_SUCCESS', payload: parsed });

      // 2. 搜索餐厅
      const result = await searchRestaurants({
        keywords: parsed.keywords,
        location: state.location,
        distance: parsed.distance,
        cuisineTypes: parsed.cuisineTypes,
        priceRange: parsed.priceRange,
      });

      if (result.restaurants.length === 0) {
        dispatch({ type: 'SET_ERROR', payload: '未找到符合条件的餐厅' });
        return;
      }

      dispatch({ type: 'SEARCH_SUCCESS', payload: result.restaurants });
    } catch (error) {
      dispatch({
        type: 'SET_ERROR',
        payload: error instanceof Error ? error.message : '搜索失败',
      });
    }
  }, [state.query, state.location, dispatch]);

  return { search, isSearching: state.isLoading };
}
```

### 3.3 转盘逻辑 Hook

```typescript
// hooks/useTurntable.ts
import { useState, useCallback, useRef } from 'react';
import { useAppContext } from '@/context/AppContext';

interface UseTurntableResult {
  rotation: number;
  isSpinning: boolean;
  spin: () => void;
}

export function useTurntable(): UseTurntableResult {
  const { state, dispatch } = useAppContext();
  const [rotation, setRotation] = useState(0);
  const [isSpinning, setIsSpinning] = useState(false);
  const spinCountRef = useRef(0);

  const spin = useCallback(() => {
    if (isSpinning || state.restaurants.length === 0) return;

    dispatch({ type: 'START_SPIN' });
    setIsSpinning(true);

    // 随机选择目标
    const targetIndex = Math.floor(Math.random() * state.restaurants.length);

    // 计算目标角度
    // 每个扇形角度 = 360 / 餐厅数量
    const segmentAngle = 360 / state.restaurants.length;

    // 目标角度 = 基础旋转(5圈) + 调整到目标扇形
    // 指针在顶部(0度)，所以目标扇形应该停在顶部
    const baseRotation = 360 * 5;
    const targetAngle = segmentAngle * targetIndex + segmentAngle / 2;
    const finalRotation = baseRotation + (360 - targetAngle);

    spinCountRef.current += 1;
    const currentSpinCount = spinCountRef.current;

    // 累加旋转角度（保持方向一致性）
    setRotation((prev) => prev + finalRotation);

    // 动画结束后更新状态
    setTimeout(() => {
      if (spinCountRef.current === currentSpinCount) {
        setIsSpinning(false);
        dispatch({ type: 'SPIN_COMPLETE', payload: targetIndex });
      }
    }, 4000); // 与 CSS 动画时长一致
  }, [isSpinning, state.restaurants, dispatch]);

  return { rotation, isSpinning, spin };
}
```

### 3.4 响应式 Hook

```typescript
// hooks/useMediaQuery.ts
import { useState, useEffect } from 'react';

export function useMediaQuery(query: string): boolean {
  const [matches, setMatches] = useState(false);

  useEffect(() => {
    const mediaQuery = window.matchMedia(query);
    setMatches(mediaQuery.matches);

    const handler = (event: MediaQueryListEvent) => {
      setMatches(event.matches);
    };

    mediaQuery.addEventListener('change', handler);
    return () => mediaQuery.removeEventListener('change', handler);
  }, [query]);

  return matches;
}

// 便捷 hooks
export function useIsMobile(): boolean {
  return useMediaQuery('(max-width: 768px)');
}

export function useIsDesktop(): boolean {
  return useMediaQuery('(min-width: 1024px)');
}
```

```typescript
  size?: number;

  const segmentAngle = useMemo(() => {

        }}

            key={restaurant.id}
            restaurant={restaurant}
            index={index}
            angle={segmentAngle}
            color={colors[index % colors.length]}
```

```typescript
import type {

const API_BASE = '/api';

async function request<T>(
  endpoint: string,
  options: RequestInit = {}
): Promise<T> {
  const response = await fetch(`${API_BASE}${endpoint}`, {
    headers: {
      'Content-Type': 'application/json',
    },
    ...options,
  });

  const data = await response.json();

  if (!data.success && data.error) {
    throw new Error(data.error.message);
  }

  return data.data;
}

// LLM 理解
```

## 六、存储工具封装

```typescript
// lib/storage.ts
import type { TurntableRecord } from '@/types';

const STORAGE_KEY = 'chisha_history';
const MAX_RECORDS = 100;

export function getHistory(): TurntableRecord[] {
  if (typeof window === 'undefined') return [];

  try {
    const data = localStorage.getItem(STORAGE_KEY);
    return data ? JSON.parse(data) : [];
  } catch {
    return [];
  }
}

export function saveRecord(record: TurntableRecord): void {
  const history = getHistory();

  // 添加到开头
  history.unshift(record);

  // 限制数量
  if (history.length > MAX_RECORDS) {
    history.splice(MAX_RECORDS);
  }

  localStorage.setItem(STORAGE_KEY, JSON.stringify(history));
}

export function deleteRecord(id: string): void {
  const history = getHistory().filter((r) => r.id !== id);
  localStorage.setItem(STORAGE_KEY, JSON.stringify(history));
}

export function clearHistory(): void {
  localStorage.removeItem(STORAGE_KEY);
}

export function createRecord(
  originalRequest: string,
  location: Location,
  selectedRestaurant: Restaurant,
  allRestaurants: Restaurant[]
): TurntableRecord {
  return {
    id: `${Date.now()}_${Math.random().toString(36).substr(2, 9)}`,
    timestamp: Date.now(),
    originalRequest,
    location,
    selectedRestaurant,
    allRestaurants,
  };
}
```
