## 二、LLM 理解接口实现

### 2.1 路由实现

```typescript
// app/api/understand/route.ts
import { NextRequest, NextResponse } from 'next/server';
import OpenAI from 'openai';

const openai = new OpenAI({
  apiKey: process.env.OPENAI_API_KEY,
  baseURL: process.env.OPENAI_BASE_URL, // 支持第三方代理端点
});

// 请求验证
interface UnderstandRequest {
  query: string;
  location: {
    lat: number;
    lng: number;
    address?: string;
  };
}

// 响应类型
interface ParsedRequirement {
  keywords: string[];
  cuisineTypes: string[];
  priceRange?: {
    min?: number;
    max?: number;
  };
  distance: number;
  requirements: string[];
}

// 系统提示词
const SYSTEM_PROMPT = `你是一个餐饮需求分析助手。用户会用自然语言描述他们想吃什么，你需要提取结构化信息。

输出 JSON 格式：
{
  "keywords": ["关键词1", "关键词2"],
  "cuisineTypes": ["菜系1", "菜系2"],
  "priceRange": { "min": 数字或null, "max": 数字或null },
  "distance": 数字（米，默认3000）,
  "requirements": ["其他需求"]
}

规则：
1. keywords: 提取用于搜索的关键词，如"火锅"、"烧烤"、"清淡"
2. cuisineTypes: 识别菜系，如"川菜"、"日料"、"西餐"
3. priceRange: 解析价格相关描述
   - "便宜"、"实惠" → max: 30
   - "不要太贵" → max: 50
   - "高档"、"贵一点" → min: 100
4. distance: 解析距离描述
   - "附近"、"近一点" → 1000
   - "周边" → 2000
   - 默认 → 3000
   - "远一点"、"不限距离" → 5000
5. requirements: 其他需求，如"有包间"、"适合约会"、"可以带宠物"

只输出 JSON，不要其他内容。`;

export async function POST(request: NextRequest) {
  try {
    const body: UnderstandRequest = await request.json();

    // 参数验证
    if (!body.query || body.query.length < 1 || body.query.length > 500) {
      return NextResponse.json(
        {
          success: false,
          error: {
            code: 'VALIDATION_ERROR',
            message: '需求内容长度应在1-500字符之间',
          },
        },
        { status: 400 }
      );
    }

    if (!body.location || typeof body.location.lat !== 'number') {
      return NextResponse.json(
        {
          success: false,
          error: {
            code: 'VALIDATION_ERROR',
            message: '位置信息无效',
          },
        },
        { status: 400 }
      );
    }

    // 构建用户消息
    const userMessage = `用户输入：${body.query}
位置：${body.location.address || `${body.location.lat}, ${body.location.lng}`}`;

    // 调用 OpenAI
    const completion = await openai.chat.completions.create({
      model: 'gpt-5.2',
      messages: [
        { role: 'system', content: SYSTEM_PROMPT },
        { role: 'user', content: userMessage },
      ],
      temperature: 0.3,
      max_tokens: 500,
      response_format: { type: 'json_object' },
    });

    const content = completion.choices[0]?.message?.content;

    if (!content) {
      throw new Error('LLM 返回为空');
    }

    // 解析 JSON
    const parsed: ParsedRequirement = JSON.parse(content);

    // 补充默认值
    const result: ParsedRequirement = {
      keywords: parsed.keywords || [],
      cuisineTypes: parsed.cuisineTypes || [],
      priceRange: parsed.priceRange,
      distance: parsed.distance || 3000,
      requirements: parsed.requirements || [],
    };

    // 如果关键词为空，使用原始输入
    if (result.keywords.length === 0) {
      result.keywords = body.query
        .replace(/[，。！？、]/g, ' ')
        .split(/\s+/)
        .filter((w) => w.length > 1)
        .slice(0, 5);
    }

    return NextResponse.json({
      success: true,
      data: result,
    });
  } catch (error) {
    console.error('Understand API Error:', error);

    // 超时或其他错误时降级处理
    if (error instanceof Error) {
      // 降级：使用简单分词
      const body: UnderstandRequest = await request.clone().json();
      const fallbackResult = fallbackParse(body.query);

      return NextResponse.json({
        success: true,
        data: fallbackResult,
        error: {
          code: 'LLM_ERROR',
          message: 'AI 理解失败，使用原始输入搜索',
        },
      });
    }

    return NextResponse.json(
      {
        success: false,
        error: {
          code: 'INTERNAL_ERROR',
          message: '服务器内部错误',
        },
      },
      { status: 500 }
    );
  }
}

// 降级解析函数
function fallbackParse(query: string): ParsedRequirement {
  const keywords = query
    .replace(/[，。！？、]/g, ' ')
    .split(/\s+/)
    .filter((w) => w.length > 1)
    .slice(0, 5);

  // 简单规则匹配
  let distance = 3000;
  if (query.includes('附近') || query.includes('近')) {
    distance = 1000;
  } else if (query.includes('远')) {
    distance = 5000;
  }

  let priceRange: { min?: number; max?: number } | undefined;
  if (query.includes('便宜') || query.includes('实惠')) {
    priceRange = { max: 30 };
  } else if (query.includes('不要太贵')) {
    priceRange = { max: 50 };
  }

  return {
    keywords,
    cuisineTypes: [],
    priceRange,
    distance,
    requirements: [],
  };
}
```

### 3.1 路由实现

```typescript
// app/api/search/route.ts
import { NextRequest, NextResponse } from 'next/server';

interface SearchRequest {
  keywords: string[];
  location: {
    lat: number;
    lng: number;
  };
  distance?: number;
  cuisineTypes?: string[];
  priceRange?: {
    min?: number;
    max?: number;
  };
}

interface Restaurant {
  id: string;
  name: string;
  cuisineType: string;
  rating?: number;
  distance: number;
  address: string;
  phone?: string;
  openingHours?: string;
  averagePrice?: number;
  location: {
    lat: number;
    lng: number;
  };
  photos?: string[];
  source: 'amap' | 'osm';
}

export async function POST(request: NextRequest) {
  try {
    const body: SearchRequest = await request.json();

    // 参数验证
    if (!body.keywords || body.keywords.length === 0) {
      return NextResponse.json(
        {
          success: false,
          error: {
            code: 'VALIDATION_ERROR',
            message: '搜索关键词不能为空',
          },
        },
        { status: 400 }
      );
    }

    if (!body.location) {
      return NextResponse.json(
        {
          success: false,
          error: {
            code: 'VALIDATION_ERROR',
            message: '位置信息无效',
          },
        },
        { status: 400 }
      );
    }

    const distance = body.distance || 3000;

    // 调用高德地图 API
    let restaurants = await searchAmap(body.keywords, body.location, distance);

    // 如果高德无结果，尝试 OSM
    if (restaurants.length === 0) {
      restaurants = await searchOSM(body.keywords, body.location, distance);
    }

    // 过滤价格范围
    if (body.priceRange) {
      restaurants = restaurants.filter((r) => {
        if (!r.averagePrice) return true;
        if (body.priceRange!.min && r.averagePrice < body.priceRange!.min) {
          return false;
        }
        if (body.priceRange!.max && r.averagePrice > body.priceRange!.max) {
          return false;
        }
        return true;
      });
    }

    // 限制返回数量
    const result = restaurants.slice(0, 8);

    if (result.length === 0) {
      return NextResponse.json({
        success: true,
        data: {
          restaurants: [],
          total: 0,
          searchParams: { keywords: body.keywords, distance },
        },
        error: {
          code: 'MAP_NO_RESULT',
          message: '未找到符合条件的餐厅',
        },
      });
    }

    return NextResponse.json({
      success: true,
      data: {
        restaurants: result,
        total: restaurants.length,
        searchParams: { keywords: body.keywords, distance },
      },
    });
  } catch (error) {
    console.error('Search API Error:', error);

    return NextResponse.json(
      {
        success: false,
        error: {
          code: 'INTERNAL_ERROR',
          message: '搜索服务暂时不可用',
        },
      },
      { status: 500 }
    );
  }
}

// 高德地图搜索
async function searchAmap(
  keywords: string[],
  location: { lat: number; lng: number },
  distance: number
): Promise<Restaurant[]> {
  const params = new URLSearchParams({
    key: process.env.AMAP_API_KEY!,
    location: `${location.lng},${location.lat}`,
    keywords: keywords.join('|'),
    types: '050000', // 餐饮服务
    radius: String(distance),
    offset: '20',
    page: '1',
    extensions: 'all',
  });

  const response = await fetch(
    `https://restapi.amap.com/v3/place/around?${params}`,
    { next: { revalidate: 300 } } // 5分钟缓存
  );

  const data = await response.json();

  if (data.status !== '1' || !data.pois) {
    return [];
  }

  return data.pois.map((poi: any) => transformAmapPoi(poi, location));
}

// 转换高德 POI 数据
function transformAmapPoi(
  poi: any,
  userLocation: { lat: number; lng: number }
): Restaurant {
  const [lng, lat] = poi.location.split(',').map(Number);

  return {
    id: `amap_${poi.id}`,
    name: poi.name,
    cuisineType: poi.type?.split(';')[0] || '餐厅',
    rating: poi.biz_ext?.rating ? parseFloat(poi.biz_ext.rating) : undefined,
    distance: parseInt(poi.distance) || 0,
    address: poi.address || '',
    phone: poi.tel || undefined,
    openingHours: poi.biz_ext?.opentime || undefined,
    averagePrice: poi.biz_ext?.cost
      ? parseInt(poi.biz_ext.cost)
      : undefined,
    location: { lat, lng },
    photos: poi.photos?.map((p: any) => p.url) || [],
    source: 'amap',
  };
}

// OSM Overpass 搜索（备选）
async function searchOSM(
  keywords: string[],
  location: { lat: number; lng: number },
  distance: number
): Promise<Restaurant[]> {
  const radiusKm = distance / 1000;

  // Overpass QL 查询
  const query = `
    [out:json][timeout:10];
    (
      node["amenity"="restaurant"](around:${distance},${location.lat},${location.lng});
      node["amenity"="cafe"](around:${distance},${location.lat},${location.lng});
      node["amenity"="fast_food"](around:${distance},${location.lat},${location.lng});
    );
    out body;
  `;

  try {
    const response = await fetch('https://overpass-api.de/api/interpreter', {
      method: 'POST',
      body: `data=${encodeURIComponent(query)}`,
      headers: {
        'Content-Type': 'application/x-www-form-urlencoded',
      },
    });

    const data = await response.json();

    if (!data.elements) {
      return [];
    }

    return data.elements
      .filter((el: any) => el.tags?.name)
      .map((el: any) => transformOsmElement(el, location))
      .slice(0, 20);
  } catch {
    return [];
  }
}

// 转换 OSM 数据
function transformOsmElement(
  element: any,
  userLocation: { lat: number; lng: number }
): Restaurant {
  const distance = calculateDistance(
    userLocation.lat,
    userLocation.lng,
    element.lat,
    element.lon
  );

  return {
    id: `osm_${element.id}`,
    name: element.tags.name,
    cuisineType: element.tags.cuisine || element.tags.amenity || '餐厅',
    rating: undefined,
    distance: Math.round(distance),
    address: element.tags['addr:full'] || element.tags['addr:street'] || '',
    phone: element.tags.phone || undefined,
    openingHours: element.tags.opening_hours || undefined,
    averagePrice: undefined,
    location: {
      lat: element.lat,
      lng: element.lon,
    },
    photos: [],
    source: 'osm',
  };
}

// 计算两点距离（米）
function calculateDistance(
  lat1: number,
  lng1: number,
  lat2: number,
  lng2: number
): number {
  const R = 6371000; // 地球半径（米）
  const dLat = ((lat2 - lat1) * Math.PI) / 180;
  const dLng = ((lng2 - lng1) * Math.PI) / 180;
  const a =
    Math.sin(dLat / 2) * Math.sin(dLat / 2) +
    Math.cos((lat1 * Math.PI) / 180) *
      Math.cos((lat2 * Math.PI) / 180) *
      Math.sin(dLng / 2) *
      Math.sin(dLng / 2);
  const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
  return R * c;
}
```

```typescript
import { NextRequest, NextResponse } from 'next/server';

interface GeocodeRequest {
  address: string;
  city?: string;
}

export async function POST(request: NextRequest) {
  try {
    const body: GeocodeRequest = await request.json();

    if (!body.address || body.address.length < 2) {
      return NextResponse.json(
        {
          success: false,
          error: {
            code: 'VALIDATION_ERROR',
            message: '地址不能为空',
          },
        },
        { status: 400 }
      );
    }

    const params = new URLSearchParams({
      key: process.env.AMAP_API_KEY!,
      address: body.address,
    });

    if (body.city) {
      params.append('city', body.city);
    }

    const response = await fetch(
      `https://restapi.amap.com/v3/geocode/geo?${params}`
    );

    const data = await response.json();

    if (data.status !== '1' || !data.geocodes || data.geocodes.length === 0) {
      return NextResponse.json({
        success: false,
        error: {
          code: 'GEOCODE_FAILED',
          message: '无法解析该地址',
        },
      });
    }

    const geocode = data.geocodes[0];
    const [lng, lat] = geocode.location.split(',').map(Number);

    return NextResponse.json({
      success: true,
      data: {
        location: { lat, lng },
        formattedAddress: geocode.formatted_address,
        city: geocode.city || geocode.province,
        district: geocode.district,
      },
    });
  } catch (error) {
    console.error('Geocode API Error:', error);

    return NextResponse.json(
      {
        success: false,
        error: {
          code: 'INTERNAL_ERROR',
          message: '地理编码服务暂时不可用',
        },
      },
      { status: 500 }
    );
  }
}
```

```typescript
// lib/withTimeout.ts
export function withTimeout<T>(
  promise: Promise<T>,
  ms: number,
  errorMessage = '请求超时'
): Promise<T> {
  const timeout = new Promise<never>((_, reject) => {
    setTimeout(() => reject(new Error(errorMessage)), ms);
  });

  return Promise.race([promise, timeout]);
}

// 使用示例
const result = await withTimeout(
  openai.chat.completions.create({ ... }),
  8000,
  'LLM 请求超时'
);
```

```typescript
    .object({
      min: z.number().optional(),
      max: z.number().optional(),
    })
    .optional(),
});

// 验证函数
export function validateRequest<T>(
  schema: z.ZodSchema<T>,
  data: unknown
): { success: true; data: T } | { success: false; error: string } {
  const result = schema.safeParse(data);

  if (result.success) {
    return { success: true, data: result.data };
  }

  return {
    success: false,
    error: result.error.errors[0]?.message || '参数验证失败',
  };
}
```

```typescript
import { NextResponse } from 'next/server';

export function successResponse<T>(data: T, meta?: { warning?: string }) {
  return NextResponse.json({
    success: true,
    data,
    ...(meta?.warning && {
      error: {
        code: 'WARNING',
        message: meta.warning,
      },
    }),
  });
}

export function errorResponse(
  code: string,
  message: string,
  status = 400
) {
  return NextResponse.json(
    {
      success: false,
      error: { code, message },
    },
    { status }
  );
}
```

## 七、日志与监控

### 7.1 简单日志工具

```typescript
// lib/logger.ts
type LogLevel = 'debug' | 'info' | 'warn' | 'error';

const LOG_LEVEL = process.env.LOG_LEVEL || 'info';
const LEVELS: Record<LogLevel, number> = {
  debug: 0,
  info: 1,
  warn: 2,
  error: 3,
};

function shouldLog(level: LogLevel): boolean {
  return LEVELS[level] >= LEVELS[LOG_LEVEL as LogLevel];
}

export const logger = {
  debug: (message: string, data?: any) => {
    if (shouldLog('debug')) {
      console.log(`[DEBUG] ${message}`, data || '');
    }
  },

  info: (message: string, data?: any) => {
    if (shouldLog('info')) {
      console.log(`[INFO] ${message}`, data || '');
    }
  },

  warn: (message: string, data?: any) => {
    if (shouldLog('warn')) {
      console.warn(`[WARN] ${message}`, data || '');
    }
  },

  error: (message: string, error?: any) => {
    if (shouldLog('error')) {
      console.error(`[ERROR] ${message}`, error || '');
    }
  },
};

// 使用示例
logger.info('Search request received', { keywords, location });
logger.error('OpenAI API failed', error);
```

### 8.3 API 限流（后期）

```typescript
// 简单限流（生产环境建议使用 Redis）
const requestCounts = new Map<string, { count: number; resetAt: number }>();

export function rateLimit(ip: string, limit = 60, windowMs = 60000): boolean {
  const now = Date.now();
  const record = requestCounts.get(ip);

  if (!record || now > record.resetAt) {
    requestCounts.set(ip, { count: 1, resetAt: now + windowMs });
    return true;
  }

  if (record.count >= limit) {
    return false;
  }

  record.count++;
  return true;
}
```
